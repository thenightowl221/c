%%
prog: stmts { printf("--- PART B: SYNTAX ANALYSIS ---\nSyntax is VALID!\n"); }
    ;

stmts: stmts stmt | /* empty */ ;

stmt: decl ';'
    | assign ';'
    | IF '(' expr ')' stmt %prec LOWER_THAN_ELSE
    | IF '(' expr ')' stmt ELSE stmt
    | WHILE '(' expr ')' stmt
    | FOR '(' forinit ';' expr ';' forstep ')' stmt
    | TYPE ID '(' ')' stmt
    | RETURN expr ';'
    | '{' stmts '}'
    ;

forinit: decl | assign | /* empty */ ;
forstep: assign | /* empty */ ;

decl: TYPE varlist ;
varlist: varlist ',' var | var ;
var: ID | ID '=' expr ;

assign: ID '=' expr | ID INC | ID DEC ;

expr: expr '+' expr | expr '-' expr | expr '*' expr | expr '/' expr
    | expr '%' expr | expr '^' expr | expr RELOP expr
    | expr AND expr | expr OR expr | NOT expr
    | '-' expr %prec UMINUS | '(' expr ')' | ID | NUM
    ;
%%

    printf("Syntax Error at line %d: %s\n", yylineno, s);
    exit(1);

%{
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

int yylex();
void yyerror(const char *s);
extern int yylineno;

int tc = 1, lc = 1;
char* new_temp() { char *t = malloc(10); sprintf(t, "t%d", tc++); return t; }
int new_lbl()    { return lc++; }
%}

%union { char *str; int lbl; }

%token TYPE IF ELSE WHILE FOR RETURN AND OR NOT INC DEC
%token <str> ID NUM RELOP
%type <str> expr
%type <lbl> ifhead

%nonassoc LOWER_THAN_ELSE
%nonassoc ELSE
%left OR
%left AND
%left RELOP
%left '+' '-'
%left '*' '/' '%'
%right NOT UMINUS
%right '^'

%%
prog: { printf("--- PART C: THREE-ADDRESS CODE ---\n"); } stmts ;

stmts: stmts stmt | /* empty */ ;

stmt: decl ';'
    | assign ';'
    | ifhead stmt %prec LOWER_THAN_ELSE { printf("L%d:\n", $1); }
    | ifhead stmt ELSE {
          int l_end = new_lbl();
          printf("goto L%d\nL%d:\n", l_end, $1);
          $<lbl>$ = l_end;
      } stmt { printf("L%d:\n", $<lbl>4); }
    | WHILE {
          int l_start = new_lbl();
          printf("L%d:\n", l_start);
          $<lbl>$ = l_start;
      } '(' expr ')' {
          int l_end = new_lbl();
          printf("ifFalse %s goto L%d\n", $4, l_end);
          $<lbl>$ = l_end;
      } stmt { printf("goto L%d\nL%d:\n", $<lbl>2, $<lbl>6); }
    | FOR '(' forinit ';' {
          int l_cond = new_lbl();
          printf("L%d:\n", l_cond);
          $<lbl>$ = l_cond;
      } expr ';' {
          int l_end  = new_lbl();
          int l_body = new_lbl();
          int l_step = new_lbl();
          printf("ifFalse %s goto L%d\ngoto L%d\nL%d:\n", $6, l_end, l_body, l_step);
          $<lbl>$ = l_end;
      } forstep ')' {
          printf("goto L%d\nL%d:\n", $<lbl>5, $<lbl>8 + 1);
      } stmt {
          printf("goto L%d\nL%d:\n", $<lbl>8 + 2, $<lbl>8);
      }
    | TYPE ID '(' ')' stmt
    | RETURN expr ';' { printf("return %s\n", $2); }
    | '{' stmts '}'
    ;

ifhead: IF '(' expr ')' {
            int l_false = new_lbl();
            printf("ifFalse %s goto L%d\n", $3, l_false);
            $$ = l_false;
        }
      ;

forinit: decl | assign | /* empty */ ;
forstep: assign | /* empty */ ;

decl: TYPE varlist ;
varlist: varlist ',' var | var ;
var: ID | ID '=' expr { printf("%s = %s\n", $1, $3); } ;

assign: ID '=' expr { printf("%s = %s\n", $1, $3); }
      | ID INC      { printf("%s = %s + 1\n", $1, $1); }
      | ID DEC      { printf("%s = %s - 1\n", $1, $1); }
      ;

expr: expr '+' expr         { $$ = new_temp(); printf("%s = %s + %s\n", $$, $1, $3); }
    | expr '-' expr         { $$ = new_temp(); printf("%s = %s - %s\n", $$, $1, $3); }
    | expr '*' expr         { $$ = new_temp(); printf("%s = %s * %s\n", $$, $1, $3); }
    | expr '/' expr         { $$ = new_temp(); printf("%s = %s / %s\n", $$, $1, $3); }

    | NOT expr              { $$ = new_temp(); printf("%s = ! %s\n", $$, $2); }
    | '-' expr %prec UMINUS { $$ = new_temp(); printf("%s = - %s\n", $$, $2); }
    | '(' expr ')'          { $$ = $2; }
    | ID                    { $$ = $1; }
    | NUM                   { $$ = $1; }
    ;

#include <stdio.h>
#include <string.h>
#include <ctype.h>

typedef struct {
    char op[15], arg1[20], arg2[20], res[20], raw[100];
    int hoisted;
} Quad;

Quad q[300];
int n = 0;

void parse_line(char *line) {
    char a[20], b[20], c[20], d[20], e[20];
    int cnt = sscanf(line, "%s %s %s %s %s", a, b, c, d, e);
    if (cnt <= 0 || line[0] == '-') return;

    strcpy(q[n].raw, line);
    q[n].hoisted = 0;
    q[n].op[0] = q[n].arg1[0] = q[n].arg2[0] = q[n].res[0] = '\0';

    if (a[strlen(a)-1] == ':') {
        strcpy(q[n].op, "label"); a[strlen(a)-1] = '\0'; strcpy(q[n].res, a);
    } else if (strcmp(a, "goto") == 0) {
        strcpy(q[n].op, "goto"); strcpy(q[n].res, b);
    } else if (strcmp(a, "ifFalse") == 0) {
        strcpy(q[n].op, "ifFalse"); strcpy(q[n].arg1, b); strcpy(q[n].res, d);
    } else if (strcmp(a, "return") == 0) {
        strcpy(q[n].op, "return"); strcpy(q[n].arg1, b);
    } else if (cnt == 3 && strcmp(b, "=") == 0) {
        strcpy(q[n].op, "="); strcpy(q[n].res, a); strcpy(q[n].arg1, c);
    } else if (cnt == 4 && strcmp(b, "=") == 0) {
        strcpy(q[n].op, c); strcpy(q[n].res, a); strcpy(q[n].arg1, d);
    } else if (cnt == 5 && strcmp(b, "=") == 0) {
        strcpy(q[n].op, d); strcpy(q[n].res, a); strcpy(q[n].arg1, c); strcpy(q[n].arg2, e);
    }
    n++;
}

int is_cond_var(char *v, int s, int e) {
    for (int i = s; i <= e; i++)
        if (strcmp(q[i].op, "ifFalse") == 0 && strcmp(q[i].arg1, v) == 0) return 1;
    return 0;
}

int is_mod(char *v, int s, int e, int self) {
    if (v[0] == '\0' || isdigit(v[0])) return 0;
    for (int i = s; i <= e; i++) {
        /* Treat the primary loop condition variable (e.g., 'm' in t2 = m < t1) as loop-varying */
        if (is_cond_var(q[i].res, s, e) && strcmp(q[i].arg1, v) == 0) return 1;
        if (i != self && !q[i].hoisted && strcmp(q[i].op, "label") != 0 &&
            strcmp(q[i].op, "goto") != 0 && strcmp(q[i].op, "ifFalse") != 0 &&
            strcmp(q[i].res, v) == 0) return 1;
    }
    return 0;
}

int main() {
    char line[100];
    while (fgets(line, sizeof(line), stdin)) {
        line[strcspn(line, "\r\n")] = 0;
        parse_line(line);
    }

    printf("--- PART D: OPTIMIZED TAC (CODE MOTION) ---\n");
    for (int i = 0; i < n; i++) {
        int end = -1;
        if (strcmp(q[i].op, "label") == 0) {
            for (int j = n - 1; j > i; j--)
                if (strcmp(q[j].op, "goto") == 0 && strcmp(q[j].res, q[i].res) == 0) { end = j; break; }
        }
        if (end != -1) {
            int chg = 1;
            while (chg) {
                chg = 0;
                for (int k = i + 1; k < end; k++) {
                    if (q[k].hoisted || q[k].res[0] == '\0') continue;
                    if (strcmp(q[k].op, "label") == 0 || strcmp(q[k].op, "goto") == 0 ||
                        strcmp(q[k].op, "ifFalse") == 0 || strcmp(q[k].op, "return") == 0) continue;
                    if (is_cond_var(q[k].res, i, end)) continue;

                    if (!is_mod(q[k].arg1, i, end, k) && !is_mod(q[k].arg2, i, end, k) && !is_mod(q[k].res, i, end, k)) {
                        q[k].hoisted = 1;
                        printf("%s\n", q[k].raw);
                        chg = 1;
                    }
                }
            }
        }
        if (!q[i].hoisted) printf("%s\n", q[i].raw);
    }
    return 0;
}


































#include <stdio.h>
#include <string.h>

int main() {
    char line[100], a[20], b[20], c[20], d[20], e[20];
    int reg = 0;

    printf("--- PART E: TARGET ASSEMBLY CODE ---\n");
    while (fgets(line, sizeof(line), stdin)) {
        line[strcspn(line, "\r\n")] = 0;
        if (line[0] == '-' || strlen(line) == 0) continue;
        int cnt = sscanf(line, "%s %s %s %s %s", a, b, c, d, e);
        if (cnt <= 0) continue;

        if (a[strlen(a)-1] == ':') {
            printf("%s\n", a);
        } else if (strcmp(a, "goto") == 0) {
            printf("\tJMP %s\n", b);
        } else if (strcmp(a, "ifFalse") == 0) {
            printf("\tCMP %s, 0\n\tJE %s\n", b, d);
        } else if (strcmp(a, "return") == 0) {
            printf("\tMOV R0, %s\n\tRET\n", b);
        } else if (cnt == 3 && strcmp(b, "=") == 0) {
            printf("\tMOV R%d, %s\n\tMOV %s, R%d\n", reg, c, a, reg);
            reg = (reg + 1) % 4;
        } else if (cnt == 4 && strcmp(b, "=") == 0) {
            printf("\tMOV R%d, %s\n", reg, d);
            printf("\t%s R%d\n", strcmp(c, "-") == 0 ? "NEG" : "NOT", reg);
            printf("\tMOV %s, R%d\n", a, reg);
            reg = (reg + 1) % 4;
        } else if (cnt == 5 && strcmp(b, "=") == 0) {
            printf("\tMOV R%d, %s\n", reg, c);
            if      (strcmp(d, "+") == 0) printf("\tADD R%d, %s\n", reg, e);
            else if (strcmp(d, "-") == 0) printf("\tSUB R%d, %s\n", reg, e);
            else if (strcmp(d, "*") == 0) printf("\tMUL R%d, %s\n", reg, e);
            else if (strcmp(d, "/") == 0) printf("\tDIV R%d, %s\n", reg, e);
            else                          printf("\tCMP R%d, %s\n", reg, e);
            printf("\tMOV %s, R%d\n", a, reg);
            reg = (reg + 1) % 4;
        }
    }
    return 0;
}

int main(int argc, char *argv[]) {

    char cmd[256];

    printf("==========================================\n");
    printf("   INTEGRATED COMPILER PIPELINE (PART F)  \n");

    system("lex part_a.l && gcc lex.yy.c -o part_a");

    system("yacc -d part_c.y && lex part_c.l && gcc y.tab.c lex.yy.c -o part_c");
    system("gcc part_d.c -o part_d");
    system("gcc part_e.c -o part_e");

    sprintf(cmd, "./part_a < %s", in); system(cmd); printf("\n");
    sprintf(cmd, "./part_b < %s", in); if (system(cmd) != 0) return 1; printf("\n");

    system("./part_d < tac.txt > opt_tac.txt && cat opt_tac.txt"); printf("\n");
    system("./part_e < opt_tac.txt");
    return 0;
}
